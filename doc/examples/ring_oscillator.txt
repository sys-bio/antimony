//------------------------------------------
//A ring oscillator (similar to the repressilator)
//The input x can be used to disturb the oscillator
//------------------------------------------
model ringoscil(x)
   gene g1, g2, g3
   species s1, s2, s3, x

   g1: -> s1; k1/(1 + x + s3^2)
   s1 -> ; k0*s1
   g2: -> s2; k1/(1 + s1^3)
   s2 -> ; k0*s2
   g3: -> s3; k1/(1 + s2^3)
   s3 -> ; k0*s3
   
   k1 = 2
   k0 = 0.1
end
